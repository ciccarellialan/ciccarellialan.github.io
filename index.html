int compute3nPlus1CycleLength(long long n) {

                  
 int cycleLength = 1;

                  
 while (n != 1) {

                  
 n = (n % 2 == 0) ? (n / 2) : (3 * n + 1);

                  
 cycleLength++;

                  
 }

                  
 return cycleLength;

                  
}
